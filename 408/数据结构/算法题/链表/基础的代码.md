# 求表长
```c
#include<stdio.h>


typedef struct LNode{
    int a;
    struct LNode *next;
}LNode,*LinkList;

int Llen(LinkList L)
{
    int len = 0;
    LNode *p = L;
    while(p ->next != NULL)
    {
        len ++;
        p = p->next;
    }
    return len;
}
int main(){
    LNode head;
    head.a = 0;
    head.next = NULL;
    
    LNode *tail = &head;
    for(int i = 0 ;i < 5 ; i ++)
    {
        LNode *p = (LNode*)malloc(sizeof(LNode));
        p->a = i * 10;
        p->next = NULL;
        tail->next = p;
        tail = p;
    }
    int t = Llen(&head);
    printf("%d",t);
}
```

```c
int LK(LinkList L,int k){
    LNode *fast = L -> next;
    LNode *slow = L-> next;
    while(fast -> next != NULL){
        fast = fast -> next;
        if(--k <= 0) slow = slow -> next;
    }
    if(slow == L) return -1;
    return slow -> a;
}
```
