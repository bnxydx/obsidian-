#include<stdio.h>
#include<stdlib.h>
typedef struct LNode{
    int data;
    struct LNode *Link;
}LNode,*LinkList;

// 建表
void build(LinkList L,int n){

    LNode *p = L;
    for(int i = 0 ;i < n; i ++)
    {

        LNode *m = (LinkList)malloc(sizeof(LNode));
        scanf("%d",&(m -> data));
        m -> Link = NULL;
        p -> Link = m;
        p = p -> Link;
    }
}
void Lprintf(LinkList L){
    LNode *p = L;
    while(p -> Link != NULL)
    {
        p = p -> Link;
        printf("%d ",p -> data);

    }
}
void check(LinkList L,int a[]){
    LNode *p = L;
    LNode *tmp;
    while(p -> Link != NULL)
    {
        tmp = p;
        p = p -> Link;
        if(a[p -> data])
            // 删除
            tmp -> Link = tmp -> Link -> Link;
        a[p -> data]++;
    }
}

int main(){
    //建表
    LNode *head = (LinkList)malloc(sizeof(LNode));
    head -> Link = NULL;
    int n = 5;
    build(head,n);
    Lprintf(head);

    int *a = (int*)malloc(sizeof(int)*100);
    check(head,a);
    printf("\n");
    Lprintf(head);
    
    return 0;
}

void reverse(LinkList L)
{
    LNode *p = L -> Link;
    LNode *tmp;
    L -> Link = NULL;
    while(p != NULL){
        tmp = p -> Link;
        p -> Link = L -> Link;
        L -> Link = p;
        p = tmp;
    }
}

一半链表反转，快慢指针，逐个插入
```c
#include <stdio.h>
#include <stdlib.h>

typedef struct Node {
    int data;
    struct Node *next;
} NODE;

// 1. 找到链表的中间结点（沿用你的快慢指针写法）
NODE *findMiddleNode(NODE *head) {
    NODE *slow = head;
    NODE *fast = head;
    while (fast != NULL) {
        fast = fast->next;
        if (fast != NULL) {
            fast = fast->next;
        }
        slow = slow->next;
    }
    return slow;
}

// 2. 反转后半段链表（沿用你的双指针写法，起点处安全置空）
NODE *reverse(NODE *start) {
    if (start == NULL) return NULL;
    NODE *p = start;
    NODE *q = start->next;
    p->next = NULL; // 关键：原起点反转后成为尾节点，必须断开置空，防止成环
    while (q != NULL) {
        NODE *tmp = q->next;
        q->next = p;
        p = q;
        q = tmp;
    }
    return p;
}

// 3. 核心解法：找中点 -> 翻转后半段 -> 穿插合并
void solve(NODE *head) {
    // 只有 0 或 1 个有效节点时无需重排
    if (head == NULL || head->next == NULL || head->next->next == NULL) {
        return;
    }

    NODE *middle = findMiddleNode(head->next); // 传入首元节点寻找中点
    NODE *list2 = reverse(middle);              // 后半段反转
    NODE *list1 = head->next;                  // 前半段起点

    // 穿插操作
    NODE *p = list1;
    NODE *q = list2;

    // 沿用你的退出条件：奇数时相遇，偶数时相邻
    while (p != q && p->next != q) {
        NODE *tmp1 = p->next;
        NODE *tmp2 = q->next;
        p->next = q;
        q->next = tmp1;
        p = tmp1;
        q = tmp2;
    }
}

// 辅助打印函数
void printList(NODE *head) {
    NODE *p = head->next;
    while (p != NULL) {
        printf("%d ", p->data);
        p = p->next;
    }
    printf("\n");
}

// 辅助建表函数（尾插法）
void build(NODE *head, int arr[], int n) {
    NODE *tail = head;
    for (int i = 0; i < n; i++) {
        NODE *m = (NODE *)malloc(sizeof(NODE));
        m->data = arr[i];
        m->next = NULL;
        tail->next = m;
        tail = m;
    }
}

int main() {
    // 测试奇数个节点：1 -> 2 -> 3 -> 4 -> 5 -> 重排为 1 -> 5 -> 2 -> 4 -> 3
    NODE *head1 = (NODE *)malloc(sizeof(NODE));
    head1->next = NULL;
    int arr1[] = {1, 2, 3, 4, 5};
    build(head1, arr1, 5);

    printf("奇数原链表: ");
    printList(head1);
    solve(head1);
    printf("重排后链表: ");
    printList(head1);

    printf("\n");

    // 测试偶数个节点：1 -> 2 -> 3 -> 4 -> 5 -> 6 -> 重排为 1 -> 6 -> 2 -> 5 -> 3 -> 4
    NODE *head2 = (NODE *)malloc(sizeof(NODE));
    head2->next = NULL;
    int arr2[] = {1, 2, 3, 4, 5, 6};
    build(head2, arr2, 6);

    printf("偶数原链表: ");
    printList(head2);
    solve(head2);
    printf("重排后链表: ");
    printList(head2);

    return 0;
}
```
