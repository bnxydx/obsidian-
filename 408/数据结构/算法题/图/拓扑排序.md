# dijkstra
```c
#include<stdio.h>
#include<stdlib.h>
#include <stdbool.h>
#include <string.h>
int N = 100;
typedef struct{
    int n,e;//n个节点e个边；
    int g[N][N];// 表
}G,*LG;
bool st[N];
void dijkstra(LG G){
    int n = G -> n;
    int dist[n + 10];
    memset(dist, 0x3f, sizeof(dist));
    memset(st, 0, sizeof(st));
    dist[1] = 0;
    for(int i = 0 ; i < n - 1 ; i ++)
    {
        int t = -1;
        for(int j = 1; j <= n ;j ++)// 从所有节点选择一个节点
            if(!st[j] && (t == -1 || dist[t] > dist[j] ))
                t = j;
        st[t] = 1;
        for(int j = 1 ;j <= n ;j ++) // 更新所有节点
            dist[j] = min(dist[j],dist[t] + G -> g[t][j]);
    }
    
}

}
```
