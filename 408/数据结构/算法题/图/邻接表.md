定义
```c
#define N 100

typedef struct ArcNode {
    int adjvex;                 // 邻接点编号
    struct ArcNode *next;       // 指向下一条边的指针
} ArcNode;

typedef struct VNode {
    int data;                   // 顶点信息
    ArcNode *first;             // 指向第一条依附该顶点的边
} VNode, AdjList[N];            // VNode 是单节点类型，AdjList 是顶点数组类型

// 在图结构体中使用：
typedef struct {
    AdjList vertices;           // 等价于 VNode vertices[N];
    int vexnum, arcnum;         // 顶点数和边数
} ALGraph;
```

纯链式定义
```c
#include <stdio.h>
#include <stdlib.h>

/* ================== 数据结构定义 ================== */

// 1. 边表节点：表示从某顶点出发的一条有向边
typedef struct ArcNode {
    int adjvex;                 // 该边所指向的邻接顶点的编号/数据
    struct ArcNode *next;       // 指向该顶点的下一条边
} ArcNode;

// 2. 顶点表节点：存储顶点信息，并串成一条顶点链表
typedef struct VNode {
    int data;                   // 顶点存储的数据（如顶点编号）
    ArcNode *first;             // 指向依附该顶点的第一条边的指针
    struct VNode *next;         // 指向图中的下一个顶点（全链表设计核心）
} VNode;

// 3. 图结构体：管理整个图的头指针及规模信息
typedef struct {
    VNode *head;                // 指向顶点链表头节点的指针
    int vexnum;                 // 图中的顶点总数
    int arcnum;                 // 图中的边总数
} ALGraph;
```
