```c
#define N 100;

typedef struct {
    int v,e;
    chat vew[N];
    int edge[N][N];
}G;

int IsExistEL(G g){
    int idx = 0;
    for(int i = 0 ;i < g.e; i ++)
    {
        int tmp = 0;
        for(int j = 0 ;j < g.e ;j ++)
            if(g[i][j])tmp++;
        
        if(tmp % 2) 
            idx ++;
        if(idx > 2) return 0;
    }
    if(idx == 0 && idx == 2) return 1;
    return 0;
}
int printVertices(G g){
    int idx = 0;
    for(int k = 0 ;k < g.e ;k ++){
        int t1 = 0;
        int t2 = 0;
        for(int i = 0 ;i < g.e ; i ++)
        {
            if(g[k][i]) t1 ++;
            if(g[i][k]) t2 ++;
        }
        if(t1 > t2){
            idx ++;
            printf("%d ",g.vew[k]);
        }
    }
}
```
