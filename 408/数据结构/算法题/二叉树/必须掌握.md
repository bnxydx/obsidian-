// 二叉树的建树 
#include<stdio.h>
#include<stdlib.h>
#include<queue>

using namespace std;
typedef struct Tree{
    int data;
    struct Tree *left,*right;
}Tree,*LinkTree;
LinkTree build(int i,int n)
{
    if(i > n) return NULL;
    
    Tree *t = (LinkTree)malloc(sizeof(Tree));
    t -> data = i;
    t -> left = build(i*2,n);
    t -> right = build(i*2 + 1,n);
    return t;
}
void proprint(LinkTree T)
{
    if(T == NULL) return ;
    
    proprint(T -> left);
    printf("%d ",T -> data);
    proprint(T -> right);
}
void floorprint(LinkTree T)
{
    queue<LinkTree> q;
    q.push(T);
    while(!q.empty()){
        Tree *t = q.front();
        q.pop();
        
        printf("%d ",t -> data);
        if(t -> left !=NULL)
            q.push(t -> left);
        if(t -> right != NULL)
            q.push(t -> right);
    }
    
}
int main(){
    int n = 10;
    Tree *root = build(1,n);
    // proprint(root);
    floorprint(root);
}

# 计算WSL
```c
int CountWPL(BiTree T, int i) {
    if (T == NULL) return 0;       // 补充返回值 0，防止空指针向下解引用崩溃
    int sum = 0;
    if (T->left == NULL && T->right == NULL)
        return T->weight * i;
    else
        sum = sum + CountWPL(T->left, i + 1) + CountWPL(T->right, i + 1);
    return sum;
}
```
